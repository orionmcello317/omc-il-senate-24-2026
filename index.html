    <!-- Script Application Logic -->

    <script>

        let myChart;


        async function updateVisualizer() {

            try {

                // Fetch the static JSON output file generated by the Python automated runner script.

                // The cache_bust query parameter prevents browsers from locking onto outdated results.

                const response = await fetch('./data/results.json?cache_bust=' + Date.now());

                const data = await response.json();


                // Convert ISO timestamp string to clean, local clock display

                const updateTime = new Date(data.last_updated).toLocaleTimeString([], { hour: '2-digit', minute: '2-digit', second: '2-digit' });

                document.getElementById('updated').innerText = `LIVE BACKEND UPDATE: ${updateTime} CST`;

                

                // 1. Update District Reporting Progress Bar Components

                document.getElementById('precinct-stats').innerText = `${data.precincts_reporting} / ${data.total_precincts} Precincts (${data.reporting_percentage}%)`;

                document.getElementById('precinct-progress-bar').style.width = `${data.reporting_percentage}%`;


                const labels = data.candidates.map(c => c.name);

                const votes = data.candidates.map(c => c.votes);


                // 2. Chart.js Render Engine Core Logic

                if (myChart) {

                    // Update layout structural inputs if the chart already exists

                    myChart.data.labels = labels;

                    myChart.data.datasets[0].data = votes;

                    myChart.update();

                } else {

                    // Initialize clean dynamic bar layout

                    const ctx = document.getElementById('resultsChart').getContext('2d');

                    myChart = new Chart(ctx, {

                        type: 'bar',

                        data: {

                            labels: labels,

                            datasets: [{

                                data: votes,

                                backgroundColor: ['#ef4444', '#3b82f6'], // Red for Republican, Blue for Democrat

                                borderRadius: 6,

                                borderWidth: 0

                            }]

                        },

                        options: {

                            indexAxis: 'y', // Makes it a horizontal bar chart

                            responsive: true,

                            maintainAspectRatio: false,

                            plugins: { 

                                legend: { display: false } 

                            },

                            scales: { 

                                y: { 

                                    grid: { display: false }, 

                                    ticks: { color: '#cbd5e1', font: { weight: 'bold', size: 13 } } 

                                }, 

                                x: { 

                                    grid: { color: '#334155' }, 

                                    ticks: { color: '#94a3b8' } 

                                } 

                            }

                        }

                    });

                }


                // 3. Render Main Total Race Row Stats Blocks

                let statsHtml = '';

                data.candidates.forEach(c => {

                    statsHtml += `

                        <div class="bg-slate-900/60 p-3 rounded-lg flex justify-between items-center border border-slate-700/40">

                            <span class="font-bold text-sm md:text-base">${c.name}</span>

                            <div class="text-right">

                                <div class="text-base md:text-lg font-mono font-bold text-blue-400">${c.votes.toLocaleString()}</div>

                                <div class="text-xs text-slate-400 font-semibold">${c.percentage}%</div>

                            </div>

                        </div>`;

                });

                document.getElementById('stats').innerHTML = statsHtml;


                // 4. Render Regional Breakdown Panels

                // DuPage County

                document.getElementById('dupage-precincts').innerText = `${data.breakdown.dupage_county.PrecinctsReporting}/${data.breakdown.dupage_county.TotalPrecincts} Rep.`;

                document.getElementById('dupage-rows').innerHTML = `

                    <div class="flex justify-between border-b border-slate-800 pb-1"><span>Seth Lewis:</span><span class="font-mono font-semibold">${data.breakdown.dupage_county.Lewis.toLocaleString()}</span></div>

                    <div class="flex justify-between pt-0.5"><span>Ben McAdams:</span><span class="font-mono font-semibold">${data.breakdown.dupage_county.McAdams.toLocaleString()}</span></div>`;


                // Cook County

                document.getElementById('cook-precincts').innerText = `${data.breakdown.cook_county.PrecinctsReporting}/${data.breakdown.cook_county.TotalPrecincts} Rep.`;

                document.getElementById('cook-rows').innerHTML = `

                    <div class="flex justify-between border-b border-slate-800 pb-1"><span>Seth Lewis:</span><span class="font-mono font-semibold">${data.breakdown.cook_county.Lewis.toLocaleString()}</span></div>

                    <div class="flex justify-between pt-0.5"><span>Ben McAdams:</span><span class="font-mono font-semibold">${data.breakdown.cook_county.McAdams.toLocaleString()}</span></div>`;


            } catch (error) {

                console.error("Visualizer drawing pipeline caught exception:", error);

            }

        }


        // Initialize and fire execution loop immediately upon landing

        updateVisualizer();

        

        // Client-side automatic fetch updates. Polls for fresh data file every 60 seconds.

        setInterval(updateVisualizer, 60000);

    </script>
